{{- /* Site overview panel for the home and list sidebars: build-time stats, a posts-per-year strip on the archives year axis (first post to the current year), how the posts split across categories, and the most used tags. It shows the blog rather than linking around it: the left sidebar already holds the page entries. It replaces the archives, categories and tag-cloud widgets. */ -}} {{- $context := .Context -}} {{- $site := $context.Site -}} {{- $pages := where $site.RegularPages "Type" "in" $site.Params.mainSections -}} {{- $notHidden := where $site.RegularPages "Params.hidden" "!=" true -}} {{- $filtered := $pages | intersect $notHidden -}} {{- $currentYear := now.Year -}} {{- $latestYear := $currentYear -}} {{- $earliestYear := $currentYear -}} {{- $maxCount := 1 -}} {{- $byYear := dict -}} {{- range $filtered.GroupByDate "2006" -}} {{- $year := int .Key -}} {{- if gt $year $latestYear }}{{ $latestYear = $year }}{{ end -}} {{- if lt $year $earliestYear }}{{ $earliestYear = $year }}{{ end -}} {{- if gt (len .Pages) $maxCount }}{{ $maxCount = len .Pages }}{{ end -}} {{- $byYear = merge $byYear (dict .Key (len .Pages)) -}} {{- end -}} {{- $archivesPage := false -}} {{- with first 1 (where $site.Pages "Layout" "archives") }}{{ $archivesPage = index . 0 }}{{ end -}} {{- /* Category shares, largest first; tones fade from the accent as on /categories/. */ -}} {{- $categories := slice -}} {{- range $site.Taxonomies.categories.ByCount -}} {{- $count := len (where .Pages "Params.hidden" "!=" true) -}} {{- if $count }}{{ $categories = $categories | append (dict "Page" .Page "Count" $count) }}{{ end -}} {{- end -}} {{- $categories = sort $categories "Count" "desc" -}} {{- $toneStep := 0.0 -}} {{- if gt (len $categories) 1 }}{{ $toneStep = div 0.65 (sub (len $categories) 1) }}{{ end -}} {{- $legendSize := 5 -}} {{- /* Most used tags, counting visible posts only. */ -}} {{- $tags := slice -}} {{- range $site.Taxonomies.tags.ByCount -}} {{- $count := len (where .Pages "Params.hidden" "!=" true) -}} {{- if $count }}{{ $tags = $tags | append (dict "Page" .Page "Count" $count) }}{{ end -}} {{- end -}} {{- $tags = sort $tags "Count" "desc" -}} {{- $tagLimit := 8 -}}
